```astro
---
import { business } from "../config/business.ts";

interface Props {
  /** Slug кейса — часть @id */
  slug: string;
  /** Название мероприятия — пойдёт в name */
  name: string;
  /** Краткое описание — пойдёт в description (берём из challenge кейса) */
  description: string;
  /** Дата мероприятия (Date) */
  startDate: Date;
  /** Имя клиента / организатора (display) */
  client: string;
  /** Площадка — пойдёт в location.name */
  venue: string;
  /** Абсолютный URL обложки (для image) */
  imageUrl: string;
}

const { slug, name, description, startDate, client, venue, imageUrl } = Astro.props;
const orgId = `${business.url}/#organization`;
const eventId = `${business.url}/cases/${slug}/#event`;

const json = {
  "@context": "https://schema.org",
  "@type": "Event",
  "@id": eventId,
  name,
  description,
  startDate: startDate.toISOString().split("T")[0],
  eventStatus: "https://schema.org/EventScheduled",
  eventAttendanceMode: "https://schema.org/OfflineEventAttendanceMode",
  location: {
    "@type": "Place",
    name: venue,
    address: {
      "@type": "PostalAddress",
      addressLocality: business.locality,
      addressRegion: business.region,
      addressCountry: business.countryCode,
    },
  },
  image: imageUrl,
  organizer: { "@id": orgId },
  about: client,
};
---
```
